cmake_minimum_required(VERSION 3.20)
project(Sync LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
option(SYNC_CUCUMBER "Run procedural Cucumber contracts" ON)
option(SYNC_BUILD_ENET_SYNC "Build the EnetSync transport adapter" ON)
include(cmake/dependencies.cmake)
option(SYNC_CHECK_CXX "Check the public C++17 interface" OFF)
if(SYNC_CHECK_CXX)
	enable_language(CXX)
endif()
set(sync_sources
	src/module.c src/server.c src/client.c src/frame.c src/unimplemented.c)
add_library(sync STATIC ${sync_sources})
add_library(Sync::sync ALIAS sync)
target_include_directories(sync PUBLIC include)
target_link_libraries(sync PUBLIC sync_flecs)
target_compile_features(sync PUBLIC c_std_11)
set_target_properties(sync PROPERTIES C_EXTENSIONS OFF)
function(sync_warnings target)
	if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
		target_compile_options(${target} PRIVATE
			-Wall -Wextra -Werror -Wpedantic)
	endif()
endfunction()
sync_warnings(sync)
if(BUILD_TESTING)
	add_subdirectory(tests)
endif()

option(SYNC_BUILD_FILE_SYNC "Build the FileSync recording adapter" ON)
if(SYNC_BUILD_FILE_SYNC)
	add_subdirectory(adapters/file_sync)
endif()

if(SYNC_BUILD_ENET_SYNC)
	add_subdirectory(adapters/enet_sync)
endif()

if(BUILD_TESTING AND SYNC_CUCUMBER)
	include(tests/cucumber/CMakeLists.txt)
endif()
