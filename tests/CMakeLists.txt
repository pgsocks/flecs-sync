set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
	"${CMAKE_CURRENT_SOURCE_DIR}/cases.def")
file(STRINGS cases.def cases REGEX "^CASE\\([a-z0-9_]+\\)$")
if(NOT cases)
	message(FATAL_ERROR "No Sync contract tests registered")
endif()
foreach(case IN LISTS cases)
	string(REGEX REPLACE "^CASE\\((.*)\\)$" "\\1" name "${case}")
	list(APPEND case_names "${name}")
	if(name MATCHES "^fault_")
		set(runner sync_fault_tests)
	else()
		set(runner sync_tests)
	endif()
	add_test(NAME "sync.${name}" COMMAND ${runner} "${name}")
	set_tests_properties("sync.${name}" PROPERTIES
		LABELS contract TIMEOUT 120)
endforeach()
# Feature IDs and executable case IDs must stay in one-to-one agreement.
file(GLOB features CONFIGURE_DEPENDS
	"${PROJECT_SOURCE_DIR}/features/core/*.feature")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${features})
foreach(feature IN LISTS features)
	file(STRINGS "${feature}" scenarios REGEX "Scenario: [a-z0-9_]+$")
	foreach(scenario IN LISTS scenarios)
		string(REGEX REPLACE ".*Scenario: " "" name "${scenario}")
		list(APPEND feature_names "${name}")
	endforeach()
endforeach()
list(SORT case_names)
list(SORT feature_names)
if(NOT "${case_names}" STREQUAL "${feature_names}")
	message(FATAL_ERROR "Feature scenarios and C case registry differ")
endif()
