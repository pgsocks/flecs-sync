foreach(runner IN ITEMS enet_sync_tests enet_sync_loopback_tests)
	add_executable(${runner} contract.c model.c)
	if(runner STREQUAL enet_sync_tests)
		target_compile_definitions(${runner} PRIVATE ENET_SYNC_TESTING)
		target_link_libraries(${runner} PRIVATE enet_sync_instrumented)
	else()
		target_link_libraries(${runner} PRIVATE EnetSync::enet_sync)
	endif()
	if(SYNC_BUILD_FILE_SYNC)
		target_compile_definitions(${runner} PRIVATE ES_FILE_SYNC)
		target_link_libraries(${runner} PRIVATE FileSync::file_sync)
	endif()
	sync_warnings(${runner})
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
	cases.def "${PROJECT_SOURCE_DIR}/features/enet_sync/enet_sync.feature")
file(STRINGS cases.def cases REGEX "^CASE\\([a-z0-9_]+\\)$")
foreach(case IN LISTS cases)
	string(REGEX REPLACE "^CASE\\((.*)\\)$" "\\1" name "${case}")
	list(APPEND case_names "${name}")
	if(name MATCHES "^recording_" AND NOT SYNC_BUILD_FILE_SYNC)
		continue()
	endif()
	file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/cases/${name}")
	add_test(NAME enet_sync.${name} COMMAND enet_sync_tests ${name})
	set_tests_properties(enet_sync.${name} PROPERTIES
		LABELS "contract;enet_sync" TIMEOUT 30 WORKING_DIRECTORY
		"${CMAKE_CURRENT_BINARY_DIR}/cases/${name}")
endforeach()
file(STRINGS
	"${PROJECT_SOURCE_DIR}/features/enet_sync/enet_sync.feature"
	scenarios REGEX "Scenario: [a-z0-9_]+$")
foreach(scenario IN LISTS scenarios)
	string(REGEX REPLACE ".*Scenario: " "" name "${scenario}")
	list(APPEND feature_names "${name}")
endforeach()
list(SORT case_names)
list(SORT feature_names)
if(NOT "${case_names}" STREQUAL "${feature_names}")
	message(FATAL_ERROR "EnetSync feature scenarios and case registry differ")
endif()
foreach(name IN ITEMS connections snapshot_delta fragmented_updates
	packet_fragmentation rpg_trade_visibility reconnect automatic_phases
	recording_replay)
	if(name MATCHES "^recording_" AND NOT SYNC_BUILD_FILE_SYNC)
		continue()
	endif()
	file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/loopback/${name}")
	add_test(NAME enet_sync.loopback_${name}
		COMMAND enet_sync_loopback_tests ${name})
	set_tests_properties(enet_sync.loopback_${name} PROPERTIES
		LABELS "loopback;enet_sync" TIMEOUT 30 WORKING_DIRECTORY
		"${CMAKE_CURRENT_BINARY_DIR}/loopback/${name}")
endforeach()
