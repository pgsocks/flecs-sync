foreach(runner IN ITEMS file_sync_rpg_tests file_sync_rpg_fault_tests)
	add_executable(${runner} main.c model.c assert.c recording.c
		gameplay.c replay.c automatic.c)
	if(runner STREQUAL file_sync_rpg_fault_tests)
		target_sources(${runner} PRIVATE failures.c)
		target_compile_definitions(${runner} PRIVATE FILE_SYNC_TESTING)
		target_include_directories(${runner} PRIVATE ..)
		target_link_libraries(${runner} PRIVATE file_sync_instrumented)
	else()
		target_link_libraries(${runner} PRIVATE FileSync::file_sync)
	endif()
	sync_warnings(${runner})
endforeach()

# The case registry names mirror stable IDs in the Gherkin specifications.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS cases.def)
file(STRINGS cases.def rpg_cases REGEX "^CASE\\(")
foreach(entry IN LISTS rpg_cases)
	string(REGEX REPLACE "^CASE\\(([a-z0-9_]+),.*" "\\1" case "${entry}")
	if(case MATCHES "^rpg_fail_")
		set(runner file_sync_rpg_fault_tests)
	else()
		set(runner file_sync_rpg_tests)
	endif()
	file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/cases/${case}")
	add_test(NAME "file_sync.${case}" COMMAND ${runner} "${case}")
	set_tests_properties("file_sync.${case}" PROPERTIES
		LABELS "contract;rpg" TIMEOUT 120 WORKING_DIRECTORY
		"${CMAKE_CURRENT_BINARY_DIR}/cases/${case}")
endforeach()

file(GLOB rpg_features CONFIGURE_DEPENDS
	"${PROJECT_SOURCE_DIR}/features/rpg/*.feature")
set(rpg_id_pattern "rpg_(record|game|replay|auto|fail)_")
string(APPEND rpg_id_pattern "[0-9][0-9][0-9][a-z0-9_]*")
foreach(feature IN LISTS rpg_features)
	set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
		"${feature}")
	file(READ "${feature}" content)
	string(REGEX MATCHALL "${rpg_id_pattern}" ids "${content}")
	list(APPEND rpg_feature_ids ${ids})
endforeach()
foreach(entry IN LISTS rpg_cases)
	string(REGEX REPLACE "^CASE\\(([a-z0-9_]+),.*" "\\1" id "${entry}")
	list(APPEND rpg_case_ids "${id}")
endforeach()
list(REMOVE_DUPLICATES rpg_feature_ids)
list(SORT rpg_feature_ids)
list(SORT rpg_case_ids)
if(NOT "${rpg_feature_ids}" STREQUAL "${rpg_case_ids}")
	message(FATAL_ERROR "RPG feature examples and C case registry differ")
endif()
